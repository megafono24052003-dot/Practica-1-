using Proyecto_biblioteca;
using System;

namespace ProyectoBiblioteca
{
    class Program
    {
        static void Main(string[] args)
        {
            Console.Write("Ingrese su nombre: ");
            string nombre = Console.ReadLine();

            BibliotecaLeonardo.Saludar(nombre);

            Console.Write("\nIngrese su edad: ");
            int edad = int.Parse(Console.ReadLine());

            BibliotecaAlex.MostrarEdad(edad);

            Console.Write("\nIngrese la nota 1: ");
            double nota1 = double.Parse(Console.ReadLine());

            Console.Write("Ingrese la nota 2: ");
            double nota2 = double.Parse(Console.ReadLine());

            BibliotecaBrandon.MostrarPromedio(nota1, nota2);

            Console.Write("\nIngrese el primer número: ");
            int num1 = int.Parse(Console.ReadLine());

            Console.Write("Ingrese el segundo número: ");
            int num2 = int.Parse(Console.ReadLine());

            BibliotecaJulian.MostrarNumeroMayor(num1, num2);

            Console.WriteLine("\nPresione una tecla para salir...");
            Console.ReadKey();
        }
    }
}
